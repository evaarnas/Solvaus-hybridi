  buildCubieCube(N) {
    const T = window.THREE;
    const total = 2.5;
    const step = total / N;
    const cubie = step * 0.95;
    this.step = step; this.half = (N - 1) / 2;
    // solid dark core so the gaps never reveal the far side
    const core = new T.Mesh(new T.BoxGeometry(total * 0.9, total * 0.9, total * 0.9), this.bodyMat);
    core.castShadow = false; core.receiveShadow = false;
    this.cubeGroup.add(core); this.core = core;
    const bodyGeo = new T.BoxGeometry(cubie, cubie, cubie);
    const stGeo = new T.PlaneGeometry(cubie * 0.84, cubie * 0.84);
    const eps = cubie * 0.5 + step * 0.012;
    const M = { px: 5, nx: 4, py: 0, ny: 1, pz: 2, nz: 3 };
    for (let ix = 0; ix < N; ix++) for (let iy = 0; iy < N; iy++) for (let iz = 0; iz < N; iz++) {
      const shell = ix === 0 || ix === N - 1 || iy === 0 || iy === N - 1 || iz === 0 || iz === N - 1;
      if (!shell) continue;
      const mesh = new T.Mesh(bodyGeo, this.bodyMat);
      const shadow = N <= 10;
      mesh.castShadow = shadow; mesh.receiveShadow = shadow;
      mesh.position.set((ix - this.half) * step, (iy - this.half) * step, (iz - this.half) * step);
      const add = (matIdx, nx, ny, nz, rotAxis, rotAng) => {
        const st = new T.Mesh(stGeo, this.faceMats[matIdx]);
        st.position.set(nx * eps, ny * eps, nz * eps);
        if (rotAxis === 'x') st.rotation.x = rotAng;
        if (rotAxis === 'y') st.rotation.y = rotAng;
        st.castShadow = false;
        mesh.add(st);
      };
      if (ix === N - 1) add(M.px, 1, 0, 0, 'y', Math.PI / 2);
      if (ix === 0) add(M.nx, -1, 0, 0, 'y', -Math.PI / 2);
      if (iy === N - 1) add(M.py, 0, 1, 0, 'x', -Math.PI / 2);
      if (iy === 0) add(M.ny, 0, -1, 0, 'x', Math.PI / 2);
      if (iz === N - 1) add(M.pz, 0, 0, 1, 'y', 0);
      if (iz === 0) add(M.nz, 0, 0, -1, 'y', Math.PI);
      mesh.userData.idx = [ix, iy, iz];
      this.cubeGroup.add(mesh);
      this.cubies.push({ mesh });
    }
    this.buildCubieLayerIndex();
  }

  buildCubieLayerIndex() {
    this.cubieLayers = Array.from({ length: 3 }, () =>
      Array.from({ length: this.N }, () => new Set())
    );
    this.cubies.forEach(c => {
      const idx = c.mesh.userData.idx;
      for (let axis = 0; axis < 3; axis++) this.cubieLayers[axis][idx[axis]].add(c);
    });
  }

  updateCubieLayerIndex(c, prevIdx, nextIdx, fixedAxis) {
    for (let axis = 0; axis < 3; axis++) {
      if (axis === fixedAxis || prevIdx[axis] === nextIdx[axis]) continue;
      this.cubieLayers[axis][prevIdx[axis]].delete(c);
      this.cubieLayers[axis][nextIdx[axis]].add(c);
    }
  }

  buildTextureCube(N) {
    const T = window.THREE;
    const total = 2.5;
    const maxTextureSize = this.renderer && this.renderer.capabilities
      ? this.renderer.capabilities.maxTextureSize : 512;
    const d = Math.min(N, maxTextureSize, 512);
    // Keep enough texels per sticker for the grid to remain legible at N=101;
    // larger faces use the direct-pixel path and need less texture detail.
    const texelsPerSticker = d <= 128 ? 10 : 4;
    const maxFaceSize = d <= 128 ? 2048 : 1024;
    const S = Math.min(maxTextureSize, maxFaceSize,
      Math.max(128, 2 ** Math.ceil(Math.log2(d * texelsPerSticker))));
    const maxAniso = this.renderer && this.renderer.capabilities
      ? this.renderer.capabilities.getMaxAnisotropy() : 4;
    const solvedIdx = [5, 4, 0, 1, 2, 3]; // px,nx,py,ny,pz,nz -> r,o,w,y,g,b
    const palette = this.faceList.map(color => [
      parseInt(color.slice(1, 3), 16),
      parseInt(color.slice(3, 5), 16),
      parseInt(color.slice(5, 7), 16),
      255,
    ]);
    this.texFaces = [];
    const mats = [];
    for (let f = 0; f < 6; f++) {
      const cv = document.createElement('canvas'); cv.width = cv.height = S;
      const ctx = cv.getContext('2d');
      const tex = new T.CanvasTexture(cv);
      tex.anisotropy = Math.min(16, maxAniso);
      tex.generateMipmaps = true;
      tex.minFilter = T.LinearMipmapLinearFilter;
      tex.magFilter = T.LinearFilter;
      const face = {
        cv, ctx, tex, d, S, palette,
        image: d > 128 ? ctx.createImageData(S, S) : null,
        solvedColor: solvedIdx[f], scr: new Uint8Array(d * d),
      };
      for (let i = 0; i < d * d; i++) face.scr[i] = Math.floor(Math.random() * this.faceList.length);
      this.texFaces.push(face);
      mats.push(new T.MeshStandardMaterial({ map: tex, roughness: 0.42, metalness: 0.0 }));
    }
    const box = new T.Mesh(new T.BoxGeometry(total, total, total), mats);
    box.castShadow = true; box.receiveShadow = true;
    this.cubeGroup.add(box); this.faceBox = box;
    this.step = total / N; this.half = (N - 1) / 2;
    for (let f = 0; f < 6; f++) this.drawFace(f, 1);
  }

  roundRect(ctx, x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.arcTo(x + w, y, x + w, y + h, r);
    ctx.arcTo(x + w, y + h, x, y + h, r);
    ctx.arcTo(x, y + h, x, y, r);
    ctx.arcTo(x, y, x + w, y, r);
    ctx.closePath();
  }

  drawFace(f, frac) {
    const face = this.texFaces[f];
    if (!face) return;
    const d = face.d, ctx = face.ctx, S = face.S, cell = S / d;
    const total = d * d, k = Math.floor(frac * total);
    if (d > 128) {
      face.synced = false;
      const pixels = face.image.data, palette = face.palette;
      for (let y = 0; y < S; y++) {
        const row = Math.min(d - 1, Math.floor(y * d / S));
        for (let x = 0; x < S; x++) {
          const rank = row * d + Math.min(d - 1, Math.floor(x * d / S));
          const color = palette[rank < k ? face.solvedColor : face.scr[rank]];
          const offset = (y * S + x) * 4;
          pixels[offset] = color[0]; pixels[offset + 1] = color[1];
          pixels[offset + 2] = color[2]; pixels[offset + 3] = 255;
        }
      }
      ctx.putImageData(face.image, 0, 0);
      face.tex.needsUpdate = true;
      return;
    }
    // The grid (dark lines with rounded sticker holes) is rendered once per face
    // into a mask; each redraw only writes d*d pixels and composites two images.
    if (!face.mask) {
      const mask = document.createElement('canvas'); mask.width = mask.height = S;
      const mctx = mask.getContext('2d');
      mctx.fillStyle = '#0a0b0d'; mctx.fillRect(0, 0, S, S);
      mctx.globalCompositeOperation = 'destination-out'; mctx.fillStyle = '#000';
      const gap = Math.max(0.5, cell * 0.09), rad = Math.min(cell * 0.2, 7);
      for (let row = 0; row < d; row++) for (let col = 0; col < d; col++) {
        const w = cell - gap * 2, h = cell - gap * 2;
        this.roundRect(mctx, col * cell + gap, row * cell + gap, w, h, Math.min(rad, w / 2)); mctx.fill();
      }
      const small = document.createElement('canvas'); small.width = small.height = d;
      face.mask = mask; face.small = small; face.sctx = small.getContext('2d');
      face.simg = face.sctx.createImageData(d, d);
    }
    const px = face.simg.data, palette = face.palette;
    for (let rank = 0; rank < total; rank++) {
      const c = palette[rank < k ? face.solvedColor : face.scr[rank]], o = rank * 4;
      px[o] = c[0]; px[o + 1] = c[1]; px[o + 2] = c[2]; px[o + 3] = 255;
    }
    face.sctx.putImageData(face.simg, 0, 0);
    ctx.imageSmoothingEnabled = false;
    ctx.drawImage(face.small, 0, 0, S, S);
    ctx.drawImage(face.mask, 0, 0);
    face.tex.needsUpdate = true;
  }

  renderTextureFromLab() {
    if (!this.lab || !this.texFaces || this.texFaces.length !== 6) return;
    const d = this.texFaces[0].d, sourceFace = [5, 4, 0, 1, 2, 3];
    const colors = this.lab.face_colors(d), area = d * d;
    if (colors.length !== 6 * area) throw new Error('invalid sticker buffer for texture display');
    this.texFaces.forEach((face, index) => {
      const offset = sourceFace[index] * area;
      face.scr = colors.slice(offset, offset + area);
      face.scr.forEach(color => {
        if (color >= this.faceList.length) {
          throw new Error('invalid sticker color for texture display');
        }
      });
    });
  }

  // Repaint only the stickers that differ from what is already on the canvas.
  // Falls back to a full redraw for small cubes or when a face's pixels are not
  // known to match face.scr (e.g. after a scramble animation drew them).
  updateTextureIncremental() {
    if (!this.lab || !this.texFaces || this.texFaces.length !== 6) return;
    const d = this.texFaces[0].d;
    if (d <= 128) {
      this.renderTextureFromLab();
      for (let f = 0; f < 6; f++) this.drawFace(f, 0);
      return;
    }
    const sourceFace = [5, 4, 0, 1, 2, 3], area = d * d;
    const colors = this.lab.face_colors(d);
    if (colors.length !== 6 * area) throw new Error('invalid sticker buffer for texture display');
    for (let f = 0; f < 6; f++) {
      const face = this.texFaces[f], off = sourceFace[f] * area;
      if (!face.synced) {
        face.scr = colors.slice(off, off + area);
        this.drawFace(f, 0);
        face.synced = true;
        continue;
      }
      const scr = face.scr, S = face.S, pixels = face.image.data, palette = face.palette;
      let minX = S, minY = S, maxX = -1, maxY = -1;
      for (let i = 0; i < area; i++) {
        const c = colors[off + i];
        if (c === scr[i]) continue;
        scr[i] = c;
        const row = (i / d) | 0, col = i - row * d;
        const x0 = Math.ceil(col * S / d), x1 = Math.ceil((col + 1) * S / d);
        const y0 = Math.ceil(row * S / d), y1 = Math.ceil((row + 1) * S / d);
        const color = palette[c] || palette[0];
        for (let y = y0; y < y1; y++) {
          let o = (y * S + x0) * 4;
          for (let x = x0; x < x1; x++, o += 4) {
            pixels[o] = color[0]; pixels[o + 1] = color[1]; pixels[o + 2] = color[2];
          }
        }
        if (x0 < minX) minX = x0;
        if (y0 < minY) minY = y0;
        if (x1 > maxX) maxX = x1;
        if (y1 > maxY) maxY = y1;
      }
      if (maxX < 0) continue;
      face.ctx.putImageData(face.image, 0, 0, minX, minY, maxX - minX, maxY - minY);
      face.tex.needsUpdate = true;
    }
  }

  textureSnapshotCount() {
    const d = this.texFaces[0].d;
    return Math.max(24, Math.min(120, Math.floor(120 * 48 * 48 / (d * d))));
  }

  // Huge native solves arrive as a packed Uint32Array (layer | axis<<16 | (turns+2)<<18)
  // so millions of moves never become individual JS objects.
  textureReplayMoveAt(i) {
    const list = this._textureReplayMoves;
    if (Array.isArray(list)) return list[i];
    if (i >= list.length) return undefined;
    const v = list[i];
    return { axis: (v >> 16) & 3, layer: v & 0xFFFF, turns: ((v >> 18) & 7) - 2 };
  }

  textureReplayTurnTotal() {
    const list = this._textureReplayMoves;
    let total = 0;
    for (let i = 0, len = list.length; i < len; i++) {
      const turns = Array.isArray(list) ? list[i].turns : ((list[i] >> 18) & 7) - 2;
      const amount = ((turns % 4) + 4) % 4;
      total += amount === 0 ? 0 : 1;
    }
    return total;
  }

  advanceTexture(dt) {
    const speedK = 0.52 / this.speedDur;
    const dur = (this.phase === 'scrambling' ? 1.3 : 2.8) / speedK;
    this.texProg = Math.min(1, this.texProg + dt / dur);
    const solving = this.phase === 'solving';
    if (solving && this._textureReplayMoves) {
      const estimated = Math.max(1, this.totalMoves);
      const target = Math.min(estimated, Math.floor(this.texProg * estimated));
      const started = performance.now();
      const initialMovesDone = this.movesDone;
      while (this.movesDone < target && performance.now() - started < 5) {
        const move = this.textureReplayMoveAt(this._textureReplayIndex);
        if (!move) break;
        const amount = ((move.turns % 4) + 4) % 4;
        if (amount === 0) {
          this._textureReplayIndex++;
          continue;
        }
        const direction = amount === 3 ? -1 : 1;
        for (let rep = amount === 2 ? 2 : 1; rep > 0; rep--) {
          this.lab.apply_design_move(move.axis, move.layer, direction);
        }
        this.movesDone++;
        this._textureReplayIndex++;
      }
      const frac = this.totalMoves ? this.movesDone / this.totalMoves : 1;
      // Redraw on a time budget, not a handful of progress buckets: huge cubes
      // otherwise sit still for seconds and then jump. The throttle scales with
      // the measured draw cost so drawing never starves the move budget.
      const nowMs = performance.now();
      const finished = this.movesDone >= this.totalMoves;
      if (this.movesDone !== initialMovesDone &&
          (finished || nowMs - (this._textureLastDraw || 0) >= Math.max(33, 3 * (this._textureDrawCost || 0)))) {
        this.updateTextureIncremental();
        this._textureLastDraw = performance.now();
        this._textureDrawCost = this._textureLastDraw - nowMs;
      }
      this.setSolvedPct(frac);
      this.ui.count.textContent = `${this.movesDone.toLocaleString()} / ${this.totalMoves.toLocaleString()}`;
      this.ui.move.style.display = 'none';
      const winnerIsGullicube = this.realWinner === 'gullicube';
      const verifiedLane = this.lanes[winnerIsGullicube ? 'beam' : 'det'];
      if (verifiedLane) {
        verifiedLane.fill.style.width = '100%';
        verifiedLane.pct.textContent = '100%';
        verifiedLane.pct.style.color = '#1BA64B';
        verifiedLane.stat.textContent = `${isTrustedResult(this) ? 'unverified' : 'verified'} · replay ${this.movesDone.toLocaleString()} / ${this.totalMoves.toLocaleString()}${this._replayElapsedLabel ? ' · elapsed ' + this._replayElapsedLabel : ''}`;
        verifiedLane.star.style.display = 'inline';
      }
      this.ui.status.textContent = isTrustedResult(this)
        ? 'Replaying ' + (winnerIsGullicube ? 'Gullicube' : 'RCube') + ' solution (replay check skipped)'
        : 'Replaying verified ' + (winnerIsGullicube ? 'Gullicube' : 'RCube') + ' solution';
      if (this._textureReplayIndex >= this._textureReplayMoves.length &&
          this.movesDone >= this.totalMoves) this.onPhaseComplete();
      return;
    }
    const frac = solving ? this.texProg : (1 - this.texProg);
    const drawBucket = Math.floor(this.texProg * this.textureSnapshotCount());
    if (drawBucket !== this._textureDrawBucket || this.texProg >= 1) {
      this._textureDrawBucket = drawBucket;
      for (let f = 0; f < 6; f++) this.drawFace(f, frac);
    }
    const est = this.totalMoves || 1;
    this.movesDone = Math.floor(this.texProg * est);
    this.setSolvedPct(frac);
    this.ui.count.textContent = `≈ ${this.movesDone.toLocaleString()} / ${est.toLocaleString()}`;
    this.ui.move.style.display = 'none';
    if (solving) { this.updateLanes(this.texProg); this.ui.status.textContent = 'Solving · reduction sweep'; }
    else this.ui.status.textContent = 'Scrambling';
    if (this.texProg >= 1) this.onPhaseComplete();
  }

  scrambleTexture() {
    this.texFaces.forEach(face => { for (let i = 0; i < face.d * face.d; i++) face.scr[i] = Math.floor(Math.random() * this.faceList.length); });
    this.phase = 'scrambling'; this.busy = true; this.scrambled = true;
    this.texProg = 0; this._textureDrawBucket = -1;
    this.totalMoves = Math.round(0.35 * this.N * this.N); this.movesDone = 0;
    this.resetLanes();
    this.ui.count.style.display = 'inline-block';
    this.syncControls();
  }

  solveTexture() {
    this.phase = 'solving'; this.busy = true; this.texProg = 0; this._textureDrawBucket = -1;
    this.totalMoves = this._textureReplayMoves
      ? this.textureReplayTurnTotal()
      : Number(this.realMoveCount) || Math.round(0.35 * this.N * this.N);
    this.movesDone = 0;
    this._textureReplayIndex = 0; this._textureReplayTurnsLeft = 0;
    this.startLanes();
    this.ui.count.style.display = 'inline-block';
    this.syncControls();
  }

  updateBadges() {
    this.ui.nBadge.textContent = `${this.N}×${this.N}`;
    const pieces = this.N === 1 ? 1 : (6 * this.N * this.N - 12 * this.N + 8);
    this.ui.pieces.textContent = `${pieces.toLocaleString()} pieces`;
    const pcsStr = pieces >= 1e9 ? (pieces / 1e9).toFixed(1) + 'B' : pieces >= 1e6 ? (pieces / 1e6).toFixed(1) + 'M' : pieces.toLocaleString();
    this.ui.nInfo.textContent = `${this.N.toLocaleString()}³ · ${pcsStr} pcs`;
  }

  // ---- moves ----
  randMove() {
    const axis = Math.floor(Math.random() * 3);
    const layer = Math.floor(Math.random() * this.N);
    const dir = Math.random() < 0.5 ? 1 : -1;
    return { axis, layer, dir };
  }

  axisVec(axis) {
    const T = window.THREE;
    return axis === 0 ? new T.Vector3(1, 0, 0) : axis === 1 ? new T.Vector3(0, 1, 0) : new T.Vector3(0, 0, 1);
  }

  // Rotate lattice indices exactly instead of rounding rotated world positions.
  // On even cubes the legal coordinates are half-step offsets; rounding them to
  // integer step multiples makes cubies drift, collide, and eventually fly apart.
  rotatedIdx(idx, m) {
    const hi = this.N - 1;
    const [x, y, z] = idx;
    if (m.axis === 0) return m.dir > 0 ? [x, hi - z, y] : [x, z, hi - y];
    if (m.axis === 1) return m.dir > 0 ? [z, y, hi - x] : [hi - z, y, x];
    return m.dir > 0 ? [hi - y, x, z] : [y, hi - x, z];
  }

  positionForIdx(idx) {
    const T = window.THREE;
    return new T.Vector3(
      (idx[0] - this.half) * this.step,
      (idx[1] - this.half) * this.step,
      (idx[2] - this.half) * this.step,
    );
  }

  startMove(m) {
    const T = window.THREE;
    const sel = Array.from(this.cubieLayers[m.axis][m.layer]);
    const pivot = new T.Group();
    this.cubeGroup.add(pivot);
    sel.forEach(c => {
      c._p0 = c.mesh.position.clone();
      c._q0 = c.mesh.quaternion.clone();
      pivot.attach(c.mesh);
    });
    const total = 2.5;
    const sizes = [total * 1.04, total * 1.04, total * 1.04];
    sizes[m.axis] = this.step * 1.06;
    const slab = new T.Mesh(new T.BoxGeometry(sizes[0], sizes[1], sizes[2]), new T.MeshBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.22, depthWrite: false, blending: T.AdditiveBlending }));
    slab.position[['x','y','z'][m.axis]] = (m.layer - this.half) * this.step;
    pivot.add(slab);
    // Per-move duration: keep the leisurely default for short (3×3) solutions, but scale
    // down for long reduction solutions so the whole replay stays ~14 s at Normal speed
    // (floored at ~one frame). Without this a 475-move 4×4 reduction solve, at 0.52 s/move,
    // would animate for 4 minutes (and an 8×8 for over half an hour).
    const _scale = Math.min(1, 26.9 / Math.max(1, this.totalMoves));
    const _dur = Math.max(0.014, this.speedDur * _scale);
    this.activeMove = { m, sel, pivot, slab, t: 0, dur: _dur, axisName: ['x','y','z'][m.axis] };
  }

  finalizeMove() {
    const T = window.THREE;
    const a = this.activeMove;
    const qMove = new T.Quaternion().setFromAxisAngle(this.axisVec(a.m.axis), a.m.dir * Math.PI / 2);
    a.sel.forEach(c => {
      const prevIdx = c.mesh.userData.idx;
      const nextIdx = this.rotatedIdx(prevIdx, a.m);
      this.updateCubieLayerIndex(c, prevIdx, nextIdx, a.m.axis);
      c.mesh.position.copy(this.positionForIdx(nextIdx));
      c.mesh.quaternion.copy(qMove.clone().multiply(c._q0));
      this.cubeGroup.add(c.mesh);
      c.mesh.userData.idx = nextIdx;
    });
    this.cubeGroup.remove(a.pivot);
    this.activeMove = null;
  }

  applyInstant(m) {
    const T = window.THREE;
    const sel = this.cubieLayers[m.axis][m.layer];
    const qMove = new T.Quaternion().setFromAxisAngle(this.axisVec(m.axis), m.dir * Math.PI / 2);
    sel.forEach(c => {
      const prevIdx = c.mesh.userData.idx;
      const nextIdx = this.rotatedIdx(prevIdx, m);
      this.updateCubieLayerIndex(c, prevIdx, nextIdx, m.axis);
      c.mesh.position.copy(this.positionForIdx(nextIdx));
      c.mesh.quaternion.premultiply(qMove);
      c.mesh.userData.idx = nextIdx;
    });
  }

  // ---- phases ----
  scramble() {
    if (this.busy) return;
    this.resetSolved(true);
    if (this.mode === 'texture') return this.scrambleTexture();
    const depth = this.scrambleDepth;
    this.lastScramble = [];
    let prev = -1;
    for (let i = 0; i < depth; i++) {
      let m = this.randMove();
      while (m.axis === prev && Math.random() < 0.6) m = this.randMove();
      prev = m.axis;
      this.lastScramble.push(m);
    }
    this.queue = this.lastScramble.slice();
    this.phase = 'scrambling'; this.busy = true; this.scrambled = true;
    this.movesDone = 0; this.totalMoves = depth;
    this.resetLanes();
    this.ui.count.style.display = 'inline-block';
    this.syncControls();
  }

  solve() {
    if (this.busy || !this.scrambled) return;
    if (this.mode === 'texture') return this.solveTexture();
    const inv = this.lastScramble.slice().reverse().map(m => ({ axis: m.axis, layer: m.layer, dir: -m.dir }));
    this.lastSolution = inv;
    this.queue = inv.slice();
    this.phase = 'solving'; this.busy = true;
    this.movesDone = 0; this.totalMoves = inv.length;
    this.startLanes();
    this.ui.count.style.display = 'inline-block';
    this.syncControls();
  }

  replay() {
    if (this.busy) return;
    if (this.mode === 'texture') {
      if (!this.texFaces) return;
      this.scrambled = true;
      for (let f = 0; f < 6; f++) this.drawFace(f, 0);
      return this.solveTexture();
    }
    if (this.lastScramble.length === 0) return;
    this.buildCube(this.n);
    this.lastScramble.forEach(m => this.applyInstant(m));
    this.scrambled = true;
    this.solve();
  }

  resetSolved(silent) {
    if (this.busy && !silent) return;
    this.buildCube(this.n);
    this.queue = []; this.activeMove = null;
    this.phase = 'idle'; this.scrambled = false; this.busy = false;
    this.movesDone = 0; this.totalMoves = 0; this.solveProgress = 1;
    if (!silent) {
      this.resetLanes();
      this.ui.count.style.display = 'none';
      this.ui.move.style.display = 'none';
      this.setSolvedPct(1);
      this.ui.status.textContent = 'Solved — scramble to begin';
      this.ui.solCount.textContent = '—';
      if (this.ui.solTime) this.ui.solTime.textContent = '';
      this.ui.solChips.innerHTML = '<span style="font-size:12.5px;color:#B7B3AA;">Solve a scramble to see the winning path.</span>';
      this.syncControls();
    }
  }

  onPhaseComplete() {
    this.ui.move.style.display = 'none';
    if (this.phase === 'scrambling') {
      this.phase = 'idle'; this.busy = false;
      this.ui.status.textContent = 'Scrambled — ready to solve';
      this.setSolvedPct(this.mode === 'texture' ? 0 : 0.04);
      this.ui.count.style.display = 'none';
    } else if (this.phase === 'solving') {
      if (this._textureReplayMoves && this.lab && !this.lab.is_solved()) {
        this.phase = 'idle'; this.busy = false;
        this.ui.status.textContent = 'Verified replay failed — cube state mismatch';
        this.ui.count.style.display = 'none';
        this.syncControls();
        return;
      }
      this.phase = 'idle'; this.busy = false; this.scrambled = false; this.hasSolution = true;
      this.setSolvedPct(1);
      this.ui.status.textContent = 'Solved ✓';
      this.ui.count.style.display = 'none';
      this.finishLanes();
      this._textureReplayMoves = null;
    }
    this.syncControls();
  }

  // ---- HUD ----
  setSolvedPct(f) {
    f = Math.max(0, Math.min(1, f));
    this.ui.ring.setAttribute('stroke-dashoffset', this.ringCirc * (1 - f));
    this.ui.pct.textContent = Math.round(f * 100);
    if (this.ui.progress) {
      const pct = Math.round(f * 100);
      this.ui.progress.style.width = pct + '%';
      this.ui.progress.setAttribute('aria-valuenow', String(pct));
    }
  }

  notation(m) {
    const N = this.N;
    const outer = m.layer === N - 1, inner0 = m.layer === 0;
    const L = ['x','y','z'][m.axis];
    let ch;
    if (m.axis === 0) ch = outer ? 'R' : inner0 ? 'L' : `${m.layer}r`;
    else if (m.axis === 1) ch = outer ? 'U' : inner0 ? 'D' : `${m.layer}u`;
    else ch = outer ? 'F' : inner0 ? 'B' : `${m.layer}f`;
    return ch + (m.dir < 0 ? "'" : '');
  }

  updateHUD() {
    const partial = this.activeMove ? Math.min(this.activeMove.t / this.activeMove.dur, 1) : 0;
    const frac = this.totalMoves ? (this.movesDone + partial) / this.totalMoves : 0;
    if (this.phase === 'scrambling') {
      this.setSolvedPct(1 - frac);
      this.ui.count.textContent = `${this.movesDone} / ${this.totalMoves}`;
      const cur = this.queue[0] || this.activeMove?.m;
      if (cur) { this.ui.status.textContent = 'Scrambling'; this.ui.move.textContent = this.notation(cur); this.ui.move.style.display = 'inline-block'; }
    } else if (this.phase === 'solving') {
      this.solveProgress = frac;
      this.setSolvedPct(frac);
      this.ui.count.textContent = `${this.movesDone} / ${this.totalMoves}`;
      this.updateLanes(frac);
      const cur = this.queue[0] || this.activeMove?.m;
      if (cur) { this.ui.status.textContent = 'Solving'; this.ui.move.textContent = this.notation(cur); this.ui.move.style.display = 'inline-block'; }
    }
  }

  // ---- lanes ----
  resetLanes() {
    Object.values(this.lanes).forEach(l => {
      l.fill.style.width = '0%'; l.pct.textContent = '0%'; l.stat.textContent = 'idle';
      l.star.style.display = 'none'; l.pct.style.color = '#A6A299';
    });
  }
  startLanes() {
    this.resetLanes();
    Object.values(this.lanes).forEach(l => l.stat.textContent = 'searching…');
  }
  updateLanes(f) {
    const small = this.n <= 4;
    const curves = small ? { det: 0.62, beam: 1.05, evo: 1.5 } : { det: 1.6, beam: 1.1, evo: 0.66 };
    const nodes = Math.round(f * (small ? 41000 : 268000));
    [['det',5800],['beam',3100],['evo',9400]].forEach(([k, mul]) => {
      const w = Math.min(1, Math.pow(f, curves[k]));
      const l = this.lanes[k];
      l.fill.style.width = (w * 100).toFixed(1) + '%';
      l.pct.textContent = Math.round(w * 100) + '%';
      l.stat.textContent = `${(nodes * (mul/5800)).toFixed(0)} nodes · gen ${Math.round(f * (k==='evo'?180:40))}`;
    });
  }
  finishLanes() {
    const small = this.n <= 4;
    const winner = small ? 'det' : 'evo';
    const moves = this.realMoveCount || this.lastSolution.length;
    Object.entries(this.lanes).forEach(([k, l]) => {
      if (k === winner) {
        l.fill.style.width = '100%'; l.pct.textContent = '100%'; l.pct.style.color = '#1BA64B';
        l.stat.textContent = `verified · ${moves} moves`; l.star.style.display = 'inline';
      } else {
        const w = 82 + Math.random() * 12;
        l.fill.style.width = w.toFixed(0) + '%'; l.pct.textContent = w.toFixed(0) + '%';
        l.stat.textContent = 'budget exhausted';
      }
    });
  }

  showSolution() {
    if (this.mode === 'texture') {
      const moveCount = this.realMoveCount || this.totalMoves;
      this.ui.solCount.textContent = moveCount.toLocaleString();
      this.ui.solChips.innerHTML = '<span style="font-size:12px;color:#8C887F;line-height:1.5;">RCube center, corner, and edge reduction — ' + moveCount.toLocaleString() + (isTrustedResult(this) ? ' moves (replay check skipped).' : ' verified moves.') + '</span>';
      return;
    }
    const moves = this.lastSolution;
    this.ui.solCount.textContent = moves.length;
    this.ui.solChips.innerHTML = '';
    moves.forEach(m => {
      const s = document.createElement('span');
      s.textContent = this.notation(m);
      s.style.cssText = "font-family:'JetBrains Mono',monospace;font-size:11.5px;padding:3px 7px;background:#F4F3EF;border-radius:6px;color:#3a3933;";
      this.ui.solChips.appendChild(s);
    });
  }

  copyPath() {
    const txt = this.lastSolution.map(m => this.notation(m)).join(' ');
    if (txt && navigator.clipboard) {
      navigator.clipboard.writeText(txt);
      const btn = this.q('[data-copy]');
      const o = btn.innerHTML;
      btn.innerHTML = '<span style="font-size:12px;color:#1BA64B;">Copied ✓</span>';
      setTimeout(() => btn.innerHTML = o, 1200);
    }
  }

  // ---- controls / settings ----
  setView(v) {
    this.view = v;
    this.ui.studio.style.display = v === 'studio' ? 'flex' : 'none';
    this.ui.studio.hidden = v !== 'studio';
    if (v === 'studio') setTimeout(() => this.resize(), 30);
    this.syncControls();
  }
  // Product sizes stay inside the desktop solver's replay-verified range. The
  // wiring layer lowers this ceiling to 5 in the standalone browser build.
  sliderToN(pos) { return Math.max(2, Math.min(this.CUBIE_MAX || 11, Math.round(pos))); }
  nToSlider(n) { return Math.max(2, Math.min(this.CUBIE_MAX || 11, Math.round(n))); }
  setN(n) {
    if (this.busy) return;
    n = Math.max(2, Math.min(this.CUBIE_MAX || 11, n));
    if (n === this.n) return;
    this.n = n;
    this.resetSolved();
  }
  setSpeed(s) {
    this.speedDur = s === 'slow' ? 0.9 : s === 'fast' ? 0.26 : 0.52;
    this.speed = s; this.syncControls();
  }

  syncControls() {
    const nsl = this.q('[data-nslider]'); if (nsl) nsl.value = this.nToSlider(this.n);
    const nv = this.q('[data-nval]'); if (nv) nv.textContent = `${this.n}×${this.n}`;
    // speed
    this.qa('[data-speed]').forEach(b => {
      const on = b.dataset.speed === (this.speed || 'normal');
      b.style.background = on ? '#fff' : 'transparent';
      b.style.color = on ? '#161514' : '#8C887F';
      b.style.boxShadow = on ? '0 1px 3px rgba(20,20,18,.1)' : 'none';
      b.setAttribute('aria-pressed', String(on));
    });
    // spin toggle
    const spin = this.q('[data-spin]');
    spin.style.color = this.autoSpin ? '#161514' : '#B7B3AA';
    spin.setAttribute('aria-pressed', String(this.autoSpin));
    // buttons enable
    const dis = (el, off) => {
      if (!el) return;
      el.disabled = !!off;
      el.setAttribute('aria-disabled', String(!!off));
      el.style.opacity = off ? '.4' : '1';
      el.style.pointerEvents = off ? 'none' : 'auto';
    };
    dis(this.q('[data-act-scramble]'), this.busy);
    dis(this.q('[data-act-solve]'), this.busy || !this.scrambled);
    dis(this.q('[data-act-replay]'), this.busy || this.lastScramble.length === 0);
    dis(this.q('[data-act-reset]'), this.busy);
    this.qa('[data-inc],[data-dec]').forEach(b => dis(b, this.busy));
  }

  // =================== LOOP ===================
  loop() {
    const T = window.THREE;
    const tick = () => {
      this._raf = requestAnimationFrame(tick);
      const dt = Math.min(this.clock.getDelta(), 0.05);

      // move animation
      if (this.mode === 'texture') {
        if (this.busy && this.phase !== 'native-scramble-pending') this.advanceTexture(dt);
      } else if (this.activeMove) {
        const a = this.activeMove;
        a.t += dt;
        const p = Math.min(a.t / a.dur, 1);
        const e = 1 - Math.pow(1 - p, 3);
        a.pivot.rotation[a.axisName] = a.m.dir * (Math.PI / 2) * e;
        if (a.t >= a.dur) {
          this.finalizeMove(); this.movesDone++;
          // Batch very long (reduction) solutions: advance extra moves instantly this frame
          // so the whole replay stays ~14 s (≈840 frames) instead of one-move-per-frame — a
          // 24k-move solve would otherwise take ~7 min even after the per-move speed-up, and
          // an 8×8 (~3.8k) ~50 s. Short 3×3 solves keep one move per frame (extra = 0).
          let extra = Math.ceil(this.totalMoves / 840) - 1;
          while (extra-- > 0 && this.queue.length) { this.applyInstant(this.queue.shift()); this.movesDone++; }
          this.updateHUD();
        }
        else this.updateHUD();
      } else if (this.queue.length) {
        this.startMove(this.queue.shift());
      } else if (this.busy) {
        this.onPhaseComplete();
      }

      // idle auto-orbit + momentum
      if (this.view === 'studio') {
        if (!this.dragging) {
          if (Math.abs(this.vel.x) > 0.0001 || Math.abs(this.vel.y) > 0.0001) {
            this.applyOrbit(this.vel.x * 0.9, this.vel.y * 0.9);
            this.vel.x *= 0.9; this.vel.y *= 0.9;
          } else if (this.autoSpin && !this.activeMove) {
            this.cubeGroup.rotation.y += 0.0024;
          }
        }
        this.renderer.render(this.scene, this.camera);
      }
    };
    tick();
  }

  componentWillUnmount() {
    if (this._raf) cancelAnimationFrame(this._raf);
    if (this.ro) this.ro.disconnect();
    if (this.onKey) window.removeEventListener('keydown', this.onKey);
  }
}
